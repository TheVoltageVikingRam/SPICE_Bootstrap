% ── Q&A — Exercise 1 ───────────────────────────────────────
\begin{frame}
    \begin{block}{Question 1}
        What is $k$?
    \end{block}
    \vspace{0.01em}
    \begin{exampleblock}{Answer}
        $K$ is the PMOS to NMOS width ratio. Since PMOS has a lower mobility than NMOS, it is inherently weaker for the same dimensions. So we use this factor to compensate for the mobility differences so as to equalize timing characteristics.
    \end{exampleblock}

    \vspace{0.01em}

    \begin{block}{Question 2}
        Why 3 dummy inverters?
    \end{block}
    \vspace{0.01em}
    \begin{exampleblock}{Answer}
        Three dummy inverters are placed before our Inverter of focus (the 4th one) so as to condition the input signal into a realistic waveform, and three after it for realistic capacitive loading. These efforts ensure that the 4th inverter operates under conditions representative of a logic gate embedded in a real logic chain.
    \end{exampleblock}
\end{frame}

% ── Part 2C: Key Insight ──────────────────────────────────
\begin{frame}{Part C — Static vs Dynamic Symmetry}
    \centering
    \vspace{0.5em}
    \begin{table}
        \centering
        \begin{tabular}{@{}lcc@{}}
            \toprule
            \textbf{Symmetry Condition} & \textbf{$k$ value} & \textbf{Type} \\
            \midrule
            $t_{PHL} = t_{PLH}$ & 1.293 & Dynamic (transient) \\
            $V_{sp} = V_{DD}/2$ & 1.503 & Static (DC) \\

            \bottomrule
        \end{tabular}
    \end{table}
    \vspace{0.8em}
    \begin{alertblock}{Key Insight}
        The $k$ for symmetric switching point ($\approx 1.503$) differs from the $k$ for symmetric propagation delay ($\approx 1.293$). The switching point \textbf{static} (DC) property, while propagation delay is a \textbf{dynamic} (transient) property; that is, they depend on different aspects of the PMOS/NMOS balance.
    \end{alertblock}
\end{frame}
